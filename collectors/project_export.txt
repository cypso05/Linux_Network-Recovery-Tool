#!/bin/bash
# diagnostics/https - HTTPS connectivity diagnostic with HTTP fallback
# Usage: check_https [url]

check_https() {
    local all_pass=true
    local https_urls=(
        "https://cloudflare.com"
        "https://google.com"
        "https://github.com"
    )
    local portal_url="http://connectivitycheck.gstatic.com/generate_204"
    
    # Allow override via argument
    if [[ -n "${1:-}" ]]; then
        https_urls=("$1")
    fi
    
    if ! command -v curl &>/dev/null; then
        echo "  âš ï¸  curl not installed - HTTPS tests skipped"
        return 1
    fi
    
    echo "  ðŸ“Œ Testing HTTPS connectivity..."
    
    local success_count=0
    local total_urls=${#https_urls[@]}
    
    # Test HTTPS targets
    for url in "${https_urls[@]}"; do
        local http_code=$(curl -s -o /dev/null -w "%{http_code}" \
            --connect-timeout 5 --max-time 10 "$url" 2>/dev/null)
        
        if [[ "$http_code" =~ ^[23][0-9][0-9]$ ]]; then
            echo "  âœ… HTTPS: $url (HTTP $http_code)"
            ((success_count++))
        elif [[ "$http_code" == "000" ]]; then
            echo "  âŒ HTTPS: $url (connection failed)"
            all_pass=false
        else
            echo "  âš ï¸  HTTPS: $url (HTTP $http_code)"
            all_pass=false
        fi
    done
    
    # Always test for captive portal (can intercept even when some HTTPS works)
    echo ""
    echo "  ðŸ“Œ Testing for captive portal..."
    local portal_code=$(curl -s -o /dev/null -w "%{http_code}" \
        --connect-timeout 5 --max-time 10 "$portal_url" 2>/dev/null)
    
    if [[ "$portal_code" == "204" ]]; then
        echo "  âœ… No captive portal detected"
    elif [[ "$portal_code" =~ ^[23][0-9][0-9]$ ]]; then
        echo "  âš ï¸  Possible captive portal (HTTP $portal_code)"
        echo "  â„¹ï¸  You may need to log in to the network"
        all_pass=false
    elif [[ "$portal_code" == "000" ]]; then
        echo "  âŒ Captive portal test: connection failed"
        all_pass=false
    else
        echo "  âš ï¸  Captive portal test: unexpected response (HTTP $portal_code)"
        all_pass=false
    fi
    
    # Summary
    if [[ $success_count -ge 1 ]]; then
        echo "  âœ… HTTPS connectivity OK ($success_count/$total_urls)"
        return 0
    else
        echo "  âŒ HTTPS connectivity FAILED (0/$total_urls)"
        return 1
    fi
}

# Usage: check_interface [iface]
# Enhanced with detailed interface information and production-ready output

check_interface() {
    local iface="${1:-$DEFAULT_IFACE}"
    local all_pass=true
    
    # Check if interface exists
    if [[ ! -d "/sys/class/net/$iface" ]]; then
        echo "  âŒ Interface $iface does not exist"

    # Gather interface details
    local operstate=$(cat "/sys/class/net/$iface/operstate" 2>/dev/null || echo "unknown")

    local duplex=$(cat "/sys/class/net/$iface/duplex" 2>/dev/null || echo "unknown")

    local iface_type="unknown"

        iface_type="wireless"
    elif [[ -d "/sys/class/net/$iface/bridge" ]]; then
        iface_type="bridge"
    elif [[ -d "/sys/class/net/$iface/bonding" ]]; then
        iface_type="bond"
    elif [[ "$iface" =~ ^(vnet|virbr) ]]; then
        iface_type="virtual"
    else
        iface_type="ethernet"
    fi
    
    # Display interface info with nice formatting
    echo "  â”Œâ”€ Interface: $iface"
    echo "  â”‚  Type: $iface_type"
    echo "  â”‚  MAC: $mac"

            echo "  â”‚  Signal: ${signal} dBm"
            echo "  â”‚  Frequency: $freq"
            ;;
        bridge)
            local slaves=$(bridge link show "$iface" 2>/dev/null | grep -oP '^[0-9]+: \K[^:]+' | tr '\n' ', ' | sed 's/, $//')
            echo "  â”‚  Slaves: ${slaves:-none}"
            ;;
        ethernet)
            [[ "$speed" != "unknown" ]] && echo "  â”‚  Speed: $speed Mb/s"
            [[ "$duplex" != "unknown" ]] && echo "  â”‚  Duplex: $duplex"
            ;;
    esac
    
    echo "  â””â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€"
    
    # Check if carrier is present (link detected)
    if [[ "$carrier" == "1" ]]; then
        echo "  âœ… Carrier: present"
    else
        if [[ "$iface_type" == "wireless" ]]; then
            echo "  âš ï¸  Carrier: no WiFi connection"
        else
            echo "  âŒ Carrier: LOST - Check cable"
        fi
        all_pass=false

        all_pass=false
    fi
    
    # Check MTU sanity
    if [[ "$mtu" != "unknown" ]] && [[ "$mtu" -lt 1280 ]]; then

        all_pass=false
    fi
    
    # Check for interface errors
    local rx_errors=$(cat "/sys/class/net/$iface/statistics/rx_errors" 2>/dev/null || echo "0")
    local tx_errors=$(cat "/sys/class/net/$iface/statistics/tx_errors" 2>/dev/null || echo "0")
    if [[ "$rx_errors" != "0" ]] || [[ "$tx_errors" != "0" ]]; then

# Usage: check_physical [iface]
# Enhanced with detailed physical layer information

    else
        iface_type="ethernet"

                echo "  â”‚  âš ï¸  Carrier: no WiFi connection"
